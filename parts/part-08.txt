<script>
(() => {
  const scenes=[...document.querySelectorAll('.scene')];
  const progress=document.getElementById('progress');
  const counter=document.getElementById('counter');
  const groupLabel=document.getElementById('groupLabel');
  const chrome=document.getElementById('chrome');
  const overview=document.getElementById('overview');
  const grid=document.getElementById('overviewGrid');
  let index=0, idleTimer=null;

  const pad=n=>String(n).padStart(2,'0');
  function hashIndex(){const id=location.hash.replace('#','');const i=scenes.findIndex(s=>s.id===id);return i>=0?i:0}
  function show(i, updateHash=true){
    index=Math.max(0,Math.min(scenes.length-1,i));
    scenes.forEach((s,n)=>s.classList.toggle('active',n===index));
    counter.textContent=`${pad(index+1)} / ${pad(scenes.length)}`;
    groupLabel.textContent=scenes[index].dataset.group||'';
    progress.style.width=`${((index+1)/scenes.length)*100}%`;
    if(updateHash) history.replaceState(null,'','#'+scenes[index].id);
    wakeChrome();
  }
  function next(){show(index+1)} function prev(){show(index-1)}
  function wakeChrome(){chrome.classList.remove('idle');clearTimeout(idleTimer);idleTimer=setTimeout(()=>chrome.classList.add('idle'),2800)}
  function openOverview(){overview.classList.add('open');document.getElementById('closeOverview').focus();wakeChrome()}
  function closeOverview(){overview.classList.remove('open')}
  function enterRead(){document.body.classList.add('read-mode');closeOverview();scenes[index].scrollIntoView({block:'start'});history.replaceState(null,'','#'+scenes[index].id);localStorage.setItem('bf-mode','read')}
  function enterPresent(){document.body.classList.remove('read-mode');show(hashIndex(),false);window.scrollTo(0,0);localStorage.setItem('bf-mode','present')}

  scenes.forEach((s,i)=>{
    const b=document.createElement('button');b.className='overview-item';b.innerHTML=`<small>${pad(i+1)} · ${s.dataset.group||''}</small><strong>${s.dataset.title}</strong><span>${i===0?'Judul dan identitas':s.querySelector('h2')?.textContent||''}</span>`;b.onclick=()=>{closeOverview();if(document.body.classList.contains('read-mode')){s.scrollIntoView({behavior:'smooth',block:'start'})}else{show(i)}};grid.appendChild(b)
  });

  document.getElementById('nextBtn').onclick=next;document.getElementById('prevBtn').onclick=prev;
  document.getElementById('overviewBtn').onclick=openOverview;document.getElementById('closeOverview').onclick=closeOverview;
  document.getElementById('readModeBtn').onclick=enterRead;document.getElementById('backToPresent').onclick=enterPresent;

  document.addEventListener('keydown',e=>{
    if(e.key.toLowerCase()==='h'&&!e.ctrlKey&&!e.metaKey&&!e.altKey){e.preventDefault();window.open('https://bad-faith-sidang.vercel.app','_blank','noopener,noreferrer');return}
    if(overview.classList.contains('open')){if(e.key==='Escape'||e.key.toLowerCase()==='o')closeOverview();return}
    if(document.body.classList.contains('read-mode')){if(e.key.toLowerCase()==='o')openOverview();return}
    if(['ArrowRight','PageDown',' '].includes(e.key)){e.preventDefault();next()}
    else if(['ArrowLeft','PageUp'].includes(e.key)){e.preventDefault();prev()}
    else if(e.key==='Home'){e.preventDefault();show(0)}
    else if(e.key==='End'){e.preventDefault();show(scenes.length-1)}
    else if(e.key.toLowerCase()==='o'){e.preventDefault();openOverview()}
    else if(e.key==='Escape'){openOverview()}
  });
  document.addEventListener('mousemove',wakeChrome,{passive:true});document.addEventListener('touchstart',wakeChrome,{passive:true});
  window.addEventListener('hashchange',()=>{if(!document.body.classList.contains('read-mode'))show(hashIndex(),false)});

  const isMobile=matchMedia('(max-width:900px)').matches;
  const saved=localStorage.getItem('bf-mode');
  index=hashIndex();
  if(saved==='read'||(!saved&&isMobile)){document.body.classList.add('read-mode');setTimeout(()=>scenes[index].scrollIntoView({block:'start'}),30)}else{show(index,false)}

})();
